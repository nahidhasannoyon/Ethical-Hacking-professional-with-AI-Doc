Extract the peh-web-labs.tar.gz file&#x20;

#### install docker&#x20;

sudo apt install docker-cli\

#### install Docker Compose&#x20;

sudo apt install docker-compose\

or&#x20;

```
sudo apt update
sudo apt upgrade
sudo apt install docker.io
sudo apt install docker-compose
```

cd labs\
sudo docker-compose up\
./set-permissions.sh\
